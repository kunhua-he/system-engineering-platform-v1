### 步骤1：设计翻译器架构

```python
# 构建装配/翻译器/老项目模块翻译.py

class 老项目模块翻译器:
    def __init__(self):
        self.号段分配表 = self.加载号段分配表()
        self.命名修复映射 = self.加载命名修复映射()
        self.翻译规范 = self.加载翻译规范()
    
    def 翻译模块(self, 老项目模块路径: str) -> dict:
        """翻译单个模块：老项目模块.json → V1契约声明JSON"""
        模块数据 = self.读取老项目模块(老项目模块路径)
        号段 = self.分配号段(模块数据["模块名"])
        契约声明 = self.生成契约声明(模块数据, 号段)
        return 契约声明
    
    def 生成契约声明(self, 模块数据: dict, 号段: dict) -> dict:
        """核心翻译逻辑"""
        能力列表 = []
        当前能力号 = 号段["号段起"]
        
        for 接口 in 模块数据["接口"]:
            能力声明 = {
                "能力编号": 当前能力号,
                "能力名": self.翻译能力名(模块数据["模块名"], 接口["能力名"]),
                "参数表": self.翻译参数表(接口["参数"]),
                "返回类型号": self.翻译类型(接口["返回"])
            }
            能力列表.append(能力声明)
            当前能力号 += 1
        
        return {
            "模块名": 模块数据["模块名"],
            "号段": f"{号段['号段起']}~{号段['号段止']}",
            "能力列表": 能力列表
        }
    
    def 翻译能力名(self, 模块名: str, 能力名: str) -> str:
        """翻译能力名：知识库 + 编目 → 知识库_编目"""
        修复后能力名 = self.修复命名(能力名)
        return f"{模块名}_{修复后能力名}"
    
    def 修复命名(self, 名字: str) -> str:
        """应用命名修复规则"""
        # 1. 确定性映射
        if 名字 in self.命名修复映射["确定性映射"]:
            return self.命名修复映射["确定性映射"][名字]
        
        # 2. 上下文映射（简化版：取第一个候选）
        if 名字 in self.命名修复映射["上下文映射"]:
            return self.命名修复映射["上下文映射"][名字]["候选"][0]
        
        # 3. 已合格，保持不变
        return 名字
    
    def 翻译参数表(self, 参数列表: list) -> list:
        """翻译参数表"""
        参数表 = []
        for 行序, 参数名 in enumerate(参数列表):
            参数表.append({
                "参数名": self.修复命名(参数名),
                "类型号": self.推断类型号(参数名),
                "行序": 行序
            })
        return 参数表
    
    def 推断类型号(self, 名字: str) -> int:
        """根据名字推断类型号（简化版）"""
        类型映射 = {
            "路径": 1,  # string
            "文本": 1,
            "编号": 2,  # int
            "数据": 10, # array
            "配置": 10,
            "结果": 20  # object
        }
        for 关键词, 类型号 in 类型映射.items():
            if 关键词 in 名字:
                return 类型号
        return 1  # 默认string
    
    def 翻译类型(self, 类型名: str) -> int:
        """PHP类型 → V1类型号"""
        类型映射 = {
            "string": 1,
            "int": 2,
            "array": 10,
            "object": 20,
            "bool": 3,
            "float": 4
        }
        return 类型映射.get(类型名, 1)
```

# 命令行用法
def main():
    import argparse
    parser = argparse.ArgumentParser(description="老项目模块翻译器")
    parser.add_argument("--模块", required=True, help="模块名（如：前端核心）")
    parser.add_argument("--输出", help="输出路径（默认：契约声明/能力声明/外部_<模块名>.json）")
    args = parser.parse_args()
    
    翻译器 = 老项目模块翻译器()
    老项目路径 = f"/Users/hekunhua/Documents/Agent/PHP/系统工程平台/{args.模块}"
    契约声明 = 翻译器.翻译模块(老项目路径)
    
    输出路径 = args.输出 or f"契约声明/能力声明/外部_{args.模块.split('_')[1]}.json"
    with open(输出路径, "w", encoding="utf-8") as f:
        json.dump(契约声明, f, ensure_ascii=False, indent=2)
    
    print(f"✓ 翻译完成：{输出路径}")

if __name__ == "__main__":
